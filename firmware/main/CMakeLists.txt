# Car profiles are embedded in the firmware image (overridable from the web app later).
file(GLOB GAUGE_PROFILE_FILES "${CMAKE_CURRENT_LIST_DIR}/../../profiles/*.json")
list(SORT GAUGE_PROFILE_FILES)
set(GAUGE_PROFILES_CPP "${CMAKE_CURRENT_BINARY_DIR}/embedded_profiles.cpp")

if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    set(decls "")
    set(entries "")
    foreach(f ${GAUGE_PROFILE_FILES})
        get_filename_component(name "${f}" NAME)
        string(MAKE_C_IDENTIFIER "${name}" sym)
        string(APPEND decls "extern const char _binary_${sym}_start[] asm(\"_binary_${sym}_start\");\n")
        string(APPEND decls "extern const char _binary_${sym}_end[] asm(\"_binary_${sym}_end\");\n")
        string(APPEND entries "    {\"${name}\", _binary_${sym}_start, _binary_${sym}_end},\n")
    endforeach()
    file(WRITE "${GAUGE_PROFILES_CPP}.in"
        "// Generated by main/CMakeLists.txt - do not edit.\n"
        "#include \"embedded_profiles.h\"\n${decls}\n"
        "const EmbeddedProfile kEmbeddedProfiles[] = {\n${entries}};\n"
        "const size_t kEmbeddedProfileCount = sizeof(kEmbeddedProfiles) / sizeof(kEmbeddedProfiles[0]);\n")
    configure_file("${GAUGE_PROFILES_CPP}.in" "${GAUGE_PROFILES_CPP}" COPYONLY)
endif()

idf_component_register(SRCS "main.cpp" "bringup_ui.cpp" "bus_task.cpp" "${GAUGE_PROFILES_CPP}"
                       INCLUDE_DIRS "."
                       EMBED_TXTFILES ${GAUGE_PROFILE_FILES}
                       REQUIRES board core gauge_hal lvgl esp_timer)
